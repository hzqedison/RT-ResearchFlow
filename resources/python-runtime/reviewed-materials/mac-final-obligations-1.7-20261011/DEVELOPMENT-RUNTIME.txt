3. RECIPIENT-SIDE MANIFEST GENERATOR
In YOUR fork create scripts/development-runtime-inventory.py with this code.
It reuses the existing inventory() and unchanged full runtime validator; it
accepts no missing/extra/changed files except the exact libquadmath paths.

import hashlib, importlib.util, json, pathlib, subprocess, sys
fork, root, baseline, target = sys.argv[1:]
fork, root, baseline = map(lambda s: pathlib.Path(s).resolve(), (fork, root, baseline))
assert baseline.parent == root.parent and root.name == 'private-python-runtime'
assert root != pathlib.Path(sys.argv[3]).resolve().parent
assert not root.is_symlink() and not (root / 'licenses/third-party/numpy/development-derivation.json').exists()
assert 'bundles' not in root.parts and '.app' not in ''.join(root.parts)
raw = baseline.read_bytes()
m = json.loads(raw)
assert target == m['platform'] + '-' + m['arch'] and target in ('darwin-arm64','darwin-x64')
spec = importlib.util.spec_from_file_location('prep', fork / 'scripts/prepare-private-python-runtime.py')
prep = importlib.util.module_from_spec(spec)
spec.loader.exec_module(prep)
rows = [r for r in prep.inventory(root) if r['path'] != 'manifest.json']
old = {r['path']: r for r in m['files']}
new = {r['path']: r for r in rows}
assert old.keys() == new.keys()
allowed = {'providers/' + p + '/site/numpy/.dylibs/libquadmath.0.dylib' for p in ('akshare','mootdx','pywencai')}
changed = [p for p in old if old[p] != new[p]]
assert changed and set(changed) <= allowed
assert all(old[p]['kind'] == new[p]['kind'] == 'file' for p in changed)
ledger = {'kind':'recipient-owned-development-derivation','finalReleaseAuthorized':False,
 'baseManifestSha256':hashlib.sha256(raw).hexdigest(),'baseSourceLockSha256':m['sourceLockSha256'],
 'target':target,'originalWheelAssetsAreInputProvenanceOnly':True,
 'generatorSha256':hashlib.sha256(pathlib.Path(__file__).read_bytes()).hexdigest(),
 'changes':[{'path':p,'before':old[p],'after':new[p]} for p in sorted(changed)]}
ledgerPath = root / 'licenses/third-party/numpy/development-derivation.json'
ledgerBytes = (json.dumps(ledger, sort_keys=True, indent=2) + '\n').encode()
ledgerPath.write_bytes(ledgerBytes)
m['sourceLockSha256'] = hashlib.sha256(ledgerBytes).hexdigest()
m['files'] = [r for r in prep.inventory(root) if r['path'] != 'manifest.json']
m['developmentDerivation'] = {'path':ledgerPath.relative_to(root).as_posix(), 'sha256':m['sourceLockSha256']}
m['finalReleaseAuthorized'] = False
payload = (json.dumps(m, sort_keys=True, indent=2) + '\n').encode()
js = "const fs=require('fs');const c=require(process.argv[1]);c.validateRuntimeTree(process.argv[2],JSON.parse(fs.readFileSync(0,'utf8')),process.argv[3]);"
subprocess.run(['node','-e',js,str(fork/'electron/shared/privatePythonRuntimeManifest.cjs'),str(root),target],input=payload,check=True)
(root / 'manifest.json').write_bytes(payload)
print('OWNED DEVELOPMENT MANIFEST SHA256', hashlib.sha256(payload).hexdigest())
